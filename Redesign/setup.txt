                inputName.addEventListener('input', (e) => {
                    previewName.textContent = e.target.value || defaultName;
                });
                inputAmt.addEventListener('input', (e) => {
                    previewAmt.textContent = '$' + (e.target.value || defaultAmt);
                });
            };

            bindTier(t1Name, t1Amount, pT1Name, pT1Amount, 'Option 1', '0');
            bindTier(t2Name, t2Amount, pT2Name, pT2Amount, 'Option 2', '0');
            bindTier(t3Name, t3Amount, pT3Name, pT3Amount, 'Option 3', '0');
        });
